## 🛠 一、闲置物品表单页（GoodsForm.vue）——新增 + 编辑

新增和编辑共用同一个组件，通过路由参数 `id` 区分模式。

```vue
<!-- src/views/goods/GoodsForm.vue -->
<script setup>
import { ref, reactive, onMounted } from 'vue'
import { useRouter, useRoute } from 'vue-router'
import { createGoods, updateGoods, getGoodsDetail } from '@/api/goods'
import { ElMessage } from 'element-plus'

const router = useRouter()
const route = useRoute()

// 判断编辑模式（有路由参数 id 就是编辑）
const isEdit = ref(!!route.params.id)

// ==================== 表单数据 ====================
const goodsForm = reactive({
  name: '',
  price: 0,
  stock: 0,
  description: '',
  image: ''      // 图片 URL 或 base64
})

// ==================== 表单验证规则 ====================
const rules = {
  name: [
    { required: true, message: '请输入闲置物品名称', trigger: 'blur' }
  ],
  price: [
    { required: true, message: '请输入价格', trigger: 'blur' },
    { type: 'number', min: 0.01, message: '价格必须大于 0', trigger: 'blur' }
  ],
  stock: [
    { required: true, message: '请输入库存', trigger: 'blur' },
    { type: 'integer', min: 0, message: '库存必须为非负整数', trigger: 'blur' }
  ]
}

// ==================== 图片上传 ====================
const imageUrl = ref('')

const handleUploadSuccess = (response) => {
  // 后端返回图片地址
  imageUrl.value = response.data.url
  goodsForm.image = response.data.url
}

const beforeUpload = (file) => {
  // 限制图片大小和类型
  const isImage = file.type.startsWith('image/')
  const isLt2M = file.size / 1024 / 1024 < 2

  if (!isImage) {
    ElMessage.error('只能上传图片文件')
    return false
  }
  if (!isLt2M) {
    ElMessage.error('图片大小不能超过 2MB')
    return false
  }
  return true
}

// ==================== 编辑模式：回填数据 ====================
onMounted(async () => {
  if (isEdit.value) {
    try {
      const res = await getGoodsDetail(route.params.id)
      const data = res.data
      goodsForm.name = data.name
      goodsForm.price = data.price
      goodsForm.stock = data.stock
      goodsForm.description = data.description || ''
      goodsForm.image = data.image || ''
      imageUrl.value = data.image || ''
    } catch (error) {
      ElMessage.error('获取闲置物品信息失败')
      router.back()
    }
  }
})

// ==================== 提交表单 ====================
const formRef = ref(null)
const submitLoading = ref(false)

const handleSubmit = async () => {
  await formRef.value.validate(async (valid) => {
    if (!valid) return

    submitLoading.value = true
    try {
      if (isEdit.value) {
        await updateGoods(route.params.id, goodsForm)
        ElMessage.success('闲置物品更新成功')
      } else {
        await createGoods(goodsForm)
        ElMessage.success('闲置物品创建成功')
      }
      router.push('/goods')
    } catch (error) {
      ElMessage.error(isEdit.value ? '更新失败' : '创建失败')
    } finally {
      submitLoading.value = false
    }
  })
}

// 取消
const handleCancel = () => {
  router.push('/goods')
}
</script>

<template>
  <div class="goods-form">
    <h2>{{ isEdit ? '编辑闲置物品' : '新增闲置物品' }}</h2>

    <el-form
      ref="formRef"
      :model="goodsForm"
      :rules="rules"
      label-width="100px"
      style="max-width: 600px"
    >
      <el-form-item label="闲置物品名称" prop="name">
        <el-input v-model="goodsForm.name" placeholder="请输入闲置物品名称" />
      </el-form-item>

      <el-form-item label="价格" prop="price">
        <el-input-number
          v-model="goodsForm.price"
          :min="0"
          :precision="2"
          :step="10"
          placeholder="请输入价格"
          style="width: 200px"
        />
      </el-form-item>

      <el-form-item label="库存" prop="stock">
        <el-input-number
          v-model="goodsForm.stock"
          :min="0"
          :step="1"
          placeholder="请输入库存"
          style="width: 200px"
        />
      </el-form-item>

      <el-form-item label="闲置物品描述">
        <el-input
          v-model="goodsForm.description"
          type="textarea"
          :rows="4"
          placeholder="请输入闲置物品描述"
        />
      </el-form-item>

      <el-form-item label="闲置物品图片">
        <el-upload
          :action="`${import.meta.env.VITE_API_BASE_URL}/upload`"
          :headers="{ Authorization: `Bearer ${localStorage.getItem('token')}` }"
          list-type="picture-card"
          :on-success="handleUploadSuccess"
          :before-upload="beforeUpload"
          :limit="1"
          :file-list="imageUrl ? [{ url: imageUrl, name: '闲置物品图片' }] : []"
        >
          <el-icon><Plus /></el-icon>
        </el-upload>
        <span class="upload-tip">支持 jpg/png 格式，大小不超过 2MB</span>
      </el-form-item>

      <el-form-item>
        <el-button
          type="primary"
          :loading="submitLoading"
          @click="handleSubmit"
        >
          {{ isEdit ? '保存修改' : '立即创建' }}
        </el-button>
        <el-button @click="handleCancel">取消</el-button>
      </el-form-item>
    </el-form>
  </div>
</template>

<style scoped>
.goods-form { padding: 20px; }
.goods-form h2 { margin-bottom: 20px; }
.upload-tip { font-size: 12px; color: #909399; margin-left: 10px; }
</style>
```

### 表单页关键设计说明

| 功能点 | 实现方式 |
|---|---|
| 新增/编辑判断 | `!!route.params.id` 判断编辑模式 |
| 编辑数据回填 | `onMounted` 里调 `getGoodsDetail` 填充表单 |
| 图片上传 | `el-upload` + `:action` 指定上传地址 + `:headers` 带 token |
| 输入验证 | `el-input-number` 限制价格/库存范围 |
| 提交 loading | `submitLoading` 防止重复提交 |

## 🔧 二、分页逻辑详解

分页是后台系统最常见的需求之一，涉及三个核心状态：

```js
const pagination = reactive({
  pageNum: 1,    // 当前页码
  pageSize: 10,  // 每页条数
  total: 0       // 总条数（后端返回）
})
```

**分页切换时的关键行为**：

| 操作 | pageNum | pageSize | 额外操作 |
|---|---|---|---|
| 切换页码 | 更新为新值 | 不变 | 重新发请求 |
| 切换每页条数 | 重置为 1 | 更新为新值 | 重新发请求 |
| 搜索 | 重置为 1 | 不变 | 重新发请求 |

> 💬 **老师备注**：切换每页条数时把 `pageNum` 重置为 1 是必须的——假设你翻到第 5 页，然后把每页条数从 10 改成 50，第 5 页可能根本不存在了。

## 🧠 本节小结

- **一套组件两用**：`isEdit = ref(!!route.params.id)` 判断新增/编辑；编辑模式在 `onMounted` 里调 `getGoodsDetail` 回填表单，回填失败 `router.back()` 兜底
- **提交分流**：`isEdit ? updateGoods(id, form) : createGoods(form)`，成功后统一 `router.push('/goods')`；`submitLoading` 防重复提交
- **表单校验规则**：名称必填、价格 `> 0`、库存非负整数；数字输入用 `el-input-number` 天然限范围
- **上传带身份**：`el-upload` 的 `:action` 指后端上传地址、`:headers` 带 `Bearer token`；`beforeUpload` 拦截非图片与超 2MB；`limit=1` + 回显 `file-list`
- **分页黄金行为**：切页码只改 `pageNum`；切每页条数要把 `pageNum` 重置为 1 再请求；搜索同理——因为"第 5 页 × 每页 50 条"可能根本不存在
